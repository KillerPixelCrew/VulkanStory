cmake_minimum_required(VERSION 3.24)
project(VulkanStoryBootstrap LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "B0's hostfxr proxy supports Windows x64 only.")
endif()
add_library(hostfxr SHARED hostfxr_proxy.cpp crash_monitor_client.cpp amd_context_compatibility.cpp hostfxr.def)
add_executable(VulkanStory.CrashMonitor crash_monitor.cpp)
target_compile_features(VulkanStory.CrashMonitor PRIVATE cxx_std_20)
target_compile_definitions(VulkanStory.CrashMonitor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(VulkanStory.CrashMonitor PRIVATE dbghelp version)
set_target_properties(VulkanStory.CrashMonitor PROPERTIES
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(hostfxr PRIVATE cxx_std_20)
target_compile_definitions(hostfxr PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(hostfxr PRIVATE advapi32 user32)
set_target_properties(hostfxr PROPERTIES PREFIX "" OUTPUT_NAME "hostfxr"
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(hostfxr PRIVATE /W4 /EHsc)
  target_compile_options(VulkanStory.CrashMonitor PRIVATE /W4 /EHsc)
else()
  target_compile_options(hostfxr PRIVATE -Wall -Wextra)
  # Also link the MinGW POSIX thread support statically; the shim must load
  # before any private native dependency resolver exists.
  target_link_options(hostfxr PRIVATE -static -static-libgcc -static-libstdc++)
  target_compile_options(VulkanStory.CrashMonitor PRIVATE -Wall -Wextra)
  target_link_options(VulkanStory.CrashMonitor PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(cet_continuation_fixture cet_continuation_fixture.cpp amd_context_compatibility.cpp crash_monitor_client.cpp)
  target_compile_features(cet_continuation_fixture PRIVATE cxx_std_20)
  target_compile_definitions(cet_continuation_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(cet_continuation_fixture PRIVATE user32 wer)
  set_target_properties(cet_continuation_fixture PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_link_options(cet_continuation_fixture PRIVATE /CETCOMPAT)
  else()
    target_link_options(cet_continuation_fixture PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--cetcompat)
  endif()
  add_executable(crash_capture_fixture crash_capture_fixture.cpp crash_monitor_client.cpp)
  target_compile_features(crash_capture_fixture PRIVATE cxx_std_20)
  target_compile_definitions(crash_capture_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(crash_capture_fixture PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(crash_capture_fixture PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_executable(fxr_version_tests fxr_version_tests.cpp)
  target_compile_features(fxr_version_tests PRIVATE cxx_std_20)
  set_target_properties(fxr_version_tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(fxr_version_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME fxr_version_order COMMAND fxr_version_tests)
endif()
