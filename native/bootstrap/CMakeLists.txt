cmake_minimum_required(VERSION 3.24)
project(VulkanStoryBootstrap LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "B0's hostfxr proxy supports Windows x64 only.")
endif()
add_library(hostfxr SHARED hostfxr_proxy.cpp hostfxr.def)
target_compile_features(hostfxr PRIVATE cxx_std_20)
target_compile_definitions(hostfxr PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(hostfxr PRIVATE advapi32)
set_target_properties(hostfxr PROPERTIES PREFIX "" OUTPUT_NAME "hostfxr"
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(hostfxr PRIVATE /W4 /EHsc)
else()
  target_compile_options(hostfxr PRIVATE -Wall -Wextra)
  # Also link the MinGW POSIX thread support statically; the shim must load
  # before any private native dependency resolver exists.
  target_link_options(hostfxr PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(fxr_version_tests fxr_version_tests.cpp)
  target_compile_features(fxr_version_tests PRIVATE cxx_std_20)
  set_target_properties(fxr_version_tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(fxr_version_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME fxr_version_order COMMAND fxr_version_tests)
endif()
